    <!-- Firebase SDK Import -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { 
            getAuth, 
            signInAnonymously, 
            signInWithCustomToken, 
            onAuthStateChanged 
        } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { 
            getFirestore, 
            collection, 
            doc, 
            addDoc, 
            setDoc,
            updateDoc,
            onSnapshot, 
            serverTimestamp 
        } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // APP ID and Configuration
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'friends-chat-app';
        
        // 独自のFirebaseプロジェクトを使用する場合は、ここをご自身のFirebaseコンソールで取得した設定に置き換えてください。
        const fallbackConfig = {
            apiKey: "AIzaSyDemoKeyForPublicTestingOnly12345",
            authDomain: "demo-chat-app.firebaseapp.com",
            projectId: "demo-chat-app",
            storageBucket: "demo-chat-app.appspot.com",
            messagingSenderId: "1234567890",
            appId: "1:1234567890:web:abcdef123456"
        };
        
        const firebaseConfig = typeof __firebase_config !== 'undefined' && __firebase_config ? JSON.parse(__firebase_config) : fallbackConfig;
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

        let db = null;
        let auth = null;
        let currentUser = null;
        let currentRoomId = null;
        let username = "";
        let selectedAvatar = "🐶";
        let unsubscribeMessages = null;
        let unsubscribeTyping = null;
        let typingTimeout = null;

        const AVATARS = ["🐶", "🐱", "🦊", "🐼", "🦁", "🐰", "🐻", "🐯", "🐨", "🐸", "🤖", "👾", "🚀", "👑", "🦄", "⚽", "🎮", "🎵"];

        const loadingScreen = document.getElementById('loading-screen');
        const authScreen = document.getElementById('auth-screen');
        const chatScreen = document.getElementById('chat-screen');
        const joinForm = document.getElementById('join-form');
        const usernameInput = document.getElementById('username-input');
        const roomInput = document.getElementById('room-input');
        const avatarSelector = document.getElementById('avatar-selector');
        const randomRoomBtn = document.getElementById('random-room-btn');
        const currentRoomName = document.getElementById('current-room-name');
        const userDisplayName = document.getElementById('user-display-name');
        const messagesContainer = document.getElementById('messages-container');
        const messageForm = document.getElementById('message-form');
        const messageInput = document.getElementById('message-input');
        const copyRoomBtn = document.getElementById('copy-room-btn');
        const leaveRoomBtn = document.getElementById('leave-room-btn');
        const typingIndicator = document.getElementById('typing-indicator');
        const typingText = document.getElementById('typing-text');

        function initAvatarSelector() {
            if (!avatarSelector) return;
            avatarSelector.innerHTML = "";
            AVATARS.forEach((emoji, index) => {
                const btn = document.createElement('button');
                btn.type = 'button';
                btn.className = `h-10 text-xl flex items-center justify-center rounded-lg border transition-all ${
                    index === 0 ? 'bg-indigo-600/30 border-indigo-500 scale-105' : 'bg-slate-900 border-slate-800 hover:bg-slate-800'
                }`;
                btn.textContent = emoji;
                btn.addEventListener('click', () => {
                    selectedAvatar = emoji;
                    Array.from(avatarSelector.children).forEach(child => {
                        child.className = 'h-10 text-xl flex items-center justify-center rounded-lg border border-slate-800 bg-slate-900 hover:bg-slate-800 transition-all';
                    });
                    btn.className = 'h-10 text-xl flex items-center justify-center rounded-lg border border-indigo-500 bg-indigo-600/30 scale-105 transition-all';
                });
                avatarSelector.appendChild(btn);
            });
        }

        async function init() {
            initAvatarSelector();
            try {
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);

                if (initialAuthToken) {
                    try {
                        await signInWithCustomToken(auth, initialAuthToken);
                    } catch (e) {
                        await signInAnonymously(auth);
                    }
                } else {
                    await signInAnonymously(auth);
                }

                onAuthStateChanged(auth, (user) => {
                    if (user) {
                        currentUser = user;
                        loadingScreen.classList.add('hidden');
                        
                        const urlParams = new URLSearchParams(window.location.search);
                        const urlRoom = urlParams.get('room');
                        if (urlRoom) {
                            roomInput.value = urlRoom;
                        }

                        authScreen.classList.remove('hidden');
                    }
                });
            } catch (err) {
                console.error("Auth init error:", err);
                loadingScreen.classList.add('hidden');
                authScreen.classList.remove('hidden');
            }
        }

        function generateRandomRoomId() {
            return 'room-' + Math.random().toString(36).substring(2, 8);
        }

        randomRoomBtn.addEventListener('click', () => {
            roomInput.value = generateRandomRoomId();
        });

        joinForm.addEventListener('submit', (e) => {
            e.preventDefault();
            const inputName = usernameInput.value.trim();
            const inputRoom = roomInput.value.trim().toLowerCase();

            if (!inputName || !inputRoom) return;

            username = inputName;
            currentRoomId = inputRoom;

            const url = new URL(window.location.href);
            url.searchParams.set('room', currentRoomId);
            window.history.pushState({}, '', url);

            authScreen.classList.add('hidden');
            chatScreen.classList.remove('hidden');

            currentRoomName.textContent = `#${currentRoomId}`;
            userDisplayName.textContent = `${selectedAvatar} ${username}`;

            listenToMessages();
            listenToTypingStatus();
        });

        function listenToMessages() {
            if (!currentUser || !currentRoomId || !db) return;

            const collectionRef = collection(db, 'artifacts', appId, 'public', 'data', `chat_${currentRoomId}`);

            if (unsubscribeMessages) unsubscribeMessages();

            unsubscribeMessages = onSnapshot(collectionRef, (snapshot) => {
                const rawDocs = [];
                snapshot.forEach(docSnap => {
                    const data = docSnap.data();
                    rawDocs.push({ id: docSnap.id, ...data });

                    if (data.userId !== currentUser.uid) {
                        const readBy = data.readBy || [];
                        if (!readBy.includes(currentUser.uid)) {
                            const msgDocRef = doc(db, 'artifacts', appId, 'public', 'data', `chat_${currentRoomId}`, docSnap.id);
                            updateDoc(msgDocRef, {
                                readBy: [...readBy, currentUser.uid]
                            }).catch(err => console.error("Error updating read status:", err));
                        }
                    }
                });

                rawDocs.sort((a, b) => {
                    const timeA = a.createdAt?.toMillis ? a.createdAt.toMillis() : (a.timestamp || 0);
                    const timeB = b.createdAt?.toMillis ? b.createdAt.toMillis() : (b.timestamp || 0);
                    return timeA - timeB;
                });

                renderMessages(rawDocs);
            }, (error) => {
                console.error("Snapshot error:", error);
                showToast("メッセージ同期エラー", "error");
            });
        }

        function renderMessages(messages) {
            messagesContainer.innerHTML = "";

            if (messages.length === 0) {
                messagesContainer.innerHTML = `
                    <div class="text-center py-16 text-slate-500">
                        <div class="inline-flex p-3 rounded-2xl bg-slate-900 border border-slate-800 mb-3 text-slate-600">
                            <i data-lucide="messages-square" class="w-8 h-8"></i>
                        </div>
                        <p class="text-sm font-medium">まだメッセージはありません</p>
                        <p class="text-xs text-slate-600 mt-1">最初のメッセージを送信してみましょう！</p>
                    </div>
                `;
                lucide.createIcons();
                return;
            }

            let lastSenderId = null;

            messages.forEach((msg) => {
                const isMe = currentUser && msg.userId === currentUser.uid;
                const dateObj = msg.createdAt?.toDate ? msg.createdAt.toDate() : (msg.timestamp ? new Date(msg.timestamp) : new Date());
                const timeStr = dateObj.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

                const readBy = msg.readBy || [];
                const isRead = isMe && currentUser && readBy.filter(uid => uid !== currentUser.uid).length > 0;

                const msgWrapper = document.createElement('div');
                msgWrapper.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'} mb-3`;

                const showName = !isMe && lastSenderId !== msg.userId;
                const avatarIcon = msg.avatar || "👤";

                let html = '';
                if (showName) {
                    html += `
                        <div class="flex items-center gap-1.5 mb-1 ml-1 text-xs">
                            <span class="text-base">${avatarIcon}</span>
                            <span class="text-indigo-400 font-medium">${escapeHtml(msg.senderName || '匿名')}</span>
                        </div>
                    `;
                }

                html += `
                    <div class="flex items-end gap-2 max-w-[85%] sm:max-w-[75%] ${isMe ? 'flex-row-reverse' : 'flex-row'}">
                        ${isMe ? `<div class="w-7 h-7 rounded-full bg-slate-800 border border-slate-700/60 flex items-center justify-center text-sm shrink-0 select-none">${selectedAvatar}</div>` : ''}
                        <div class="px-4 py-2.5 rounded-2xl ${isMe ? 'bg-indigo-600 text-white rounded-br-xs shadow-md shadow-indigo-600/10' : 'bg-slate-800 text-slate-100 rounded-bl-xs border border-slate-700/60'} text-sm leading-relaxed break-words">
                            ${escapeHtml(msg.text)}
                        </div>
                        <div class="flex flex-col ${isMe ? 'items-end' : 'items-start'} text-[10px] text-slate-500 shrink-0 select-none pb-0.5">
                            ${isRead ? '<span class="text-indigo-400 font-semibold leading-none mb-0.5">既読</span>' : ''}
                            <span class="leading-none">${timeStr}</span>
                        </div>
                    </div>
                `;

                msgWrapper.innerHTML = html;
                messagesContainer.appendChild(msgWrapper);

                lastSenderId = msg.userId;
            });

            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            lucide.createIcons();
        }

        messageForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const text = messageInput.value.trim();
            if (!text || !currentRoomId) return;

            messageInput.value = "";
            setTypingStatus(false);

            if (!db || !currentUser) return;

            try {
                const collectionRef = collection(db, 'artifacts', appId, 'public', 'data', `chat_${currentRoomId}`);
                await addDoc(collectionRef, {
                    text: text,
                    userId: currentUser.uid,
                    senderName: username,
                    avatar: selectedAvatar,
                    createdAt: serverTimestamp(),
                    timestamp: Date.now(),
                    readBy: [currentUser.uid]
                });
            } catch (err) {
                console.error("Send message error:", err);
                showToast("送信に失敗しました", "error");
            }
        });

        messageInput.addEventListener('input', () => {
            if (!currentRoomId || !currentUser) return;

            setTypingStatus(true);

            if (typingTimeout) clearTimeout(typingTimeout);
            typingTimeout = setTimeout(() => {
                setTypingStatus(false);
            }, 2000);
        });

        async function setTypingStatus(isTyping) {
            if (!currentRoomId || !currentUser || !db) return;
            const typingDocRef = doc(db, 'artifacts', appId, 'public', 'data', `typing_${currentRoomId}`, currentUser.uid);
            try {
                await setDoc(typingDocRef, {
                    username: username,
                    avatar: selectedAvatar,
                    isTyping: isTyping,
                    updatedAt: serverTimestamp()
                });
            } catch (err) {
                console.error("Typing status update error:", err);
            }
        }

        function listenToTypingStatus() {
            if (!currentRoomId || !currentUser || !db) return;

            const collectionRef = collection(db, 'artifacts', appId, 'public', 'data', `typing_${currentRoomId}`);

            if (unsubscribeTyping) unsubscribeTyping();

            unsubscribeTyping = onSnapshot(collectionRef, (snapshot) => {
                const typingUsers = [];
                snapshot.forEach((docSnap) => {
                    const data = docSnap.data();
                    if (docSnap.id !== currentUser.uid && data.isTyping) {
                        typingUsers.push(`${data.avatar || ''} ${data.username || "誰か"}`);
                    }
                });

                if (typingUsers.length > 0) {
                    const text = typingUsers.length === 1 
                        ? `${typingUsers[0]} が入力中...`
                        : `${typingUsers.join(', ')} が入力中...`;
                    typingText.textContent = text;
                    typingIndicator.classList.remove('opacity-0');
                    typingIndicator.classList.add('opacity-100');
                } else {
                    typingIndicator.classList.remove('opacity-100');
                    typingIndicator.classList.add('opacity-0');
                }
            }, (err) => console.error("Typing snapshot error:", err));
        }

        copyRoomBtn.addEventListener('click', () => {
            if (!currentRoomId) return;

            const currentUrl = window.location.href;
            const tempInput = document.createElement('input');
            tempInput.value = currentUrl;
            document.body.appendChild(tempInput);
            tempInput.select();
            try {
                document.execCommand('copy');
                showToast(`招待リンクをコピーしました！`);
            } catch (err) {
                showToast("コピーに失敗しました", "error");
            }
            document.body.removeChild(tempInput);
        });

        leaveRoomBtn.addEventListener('click', () => {
            if (unsubscribeMessages) unsubscribeMessages();
            if (unsubscribeTyping) unsubscribeTyping();
            setTypingStatus(false);
            currentRoomId = null;

            const url = new URL(window.location.href);
            url.searchParams.delete('room');
            window.history.pushState({}, '', url);

            chatScreen.classList.add('hidden');
            authScreen.classList.remove('hidden');
            messagesContainer.innerHTML = "";
            showToast("退室しました");
        });

        function showToast(message, type = "info") {
            const toast = document.getElementById('toast');
            const toastMessage = document.getElementById('toast-message');
            const toastIcon = document.getElementById('toast-icon');

            toastMessage.textContent = message;
            
            if (type === "error") {
                toastIcon.setAttribute('data-lucide', 'alert-circle');
                toastIcon.className = "w-5 h-5 text-red-400";
            } else {
                toastIcon.setAttribute('data-lucide', 'check-circle');

        function escapeHtml(str) {
            if (!str) return '';
            return str
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
